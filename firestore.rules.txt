GB STUDY — Firestore Security Rules starter

IMPORTANT: Replace YOUR_ADMIN_FIREBASE_UID with the UID of the admin account you create in Firebase Authentication. Never use a client-side password as the security boundary.

rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() {
      return request.auth != null && request.auth.token.email_verified == true;
    }
    function isAdmin() {
      return signedIn() && request.auth.uid == "YOUR_ADMIN_FIREBASE_UID";
    }

    // Public platform list can be read by guests; only the admin UID can publish changes.
    match /appData/public {
      allow read: if true;
      allow create, update, delete: if isAdmin();
    }

    // Only verified accounts can read/send chat messages. Users may only send as themselves.
    match /chatMessages/{messageId} {
      allow read: if signedIn();
      allow create: if (isAdmin() && request.resource.data.admin == true) || (signedIn() && request.resource.data.admin == false
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.email == request.auth.token.email
        && request.resource.data.text is string
        && request.resource.data.text.size() > 0
        && request.resource.data.text.size() <= 500);
      allow update, delete: if isAdmin();
    }


    // Cloud study materials: public read, admin-only writes.
    match /materials/{materialId} {
      allow read: if true;
      allow create: if isAdmin()
        && request.resource.data.createdByUid == request.auth.uid
        && request.resource.data.title is string
        && request.resource.data.title.size() > 0
        && request.resource.data.title.size() <= 100
        && request.resource.data.category is string
        && request.resource.data.category.size() > 0
        && request.resource.data.category.size() <= 60
        && request.resource.data.link is string
        && request.resource.data.link.matches('https://.*');
      allow update, delete: if isAdmin();
    }

    // Deny everything else by default.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
